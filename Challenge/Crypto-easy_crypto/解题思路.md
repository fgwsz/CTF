编写`solve.cpp`脚本,
将二进制串的内容按照空格分开并打印十进制数和可读`ASCII`序列.
```cpp
#include<cstddef>
#include<cctype>
#include<string>
#include<vector>
#include<stdexcept>
#include<iostream>

// x^y
inline unsigned char power(unsigned char x,unsigned char y){
    if(x==0){
        return 0;
    }
    if(y==0){
        return 1;
    }
    unsigned char ret=1;
    for(unsigned char i=0;i<y;++i){
        ret*=x;
    }
    return ret;
};

//将二进制字符串转化为一个unsigned char
inline unsigned char bstr_to_uchar(char const* str,std::size_t size){
    if(str==nullptr){
        throw std::runtime_error("bstr_to_uchar:str!=nullptr");
    }
    if(size==0){
        throw std::runtime_error("bstr_to_uchar:size!=0");
    }
    //x^y
    //size:             4
    //index:0, 1, 2, 3
    //value:1  1  0  1
    //      ^  4-0-1
    unsigned char ret=0;

    for(std::size_t index=0;index<size;++index){
        if(str[index]=='1'){
            ret+=power(2,size-1-index);
        }else if(str[index]=='0'){
            continue;
        }else{
            throw std::runtime_error("bstr_to_uchar:error bchar");
        }
    }
    return ret;
}

inline std::vector<unsigned char> bstr_to_uchar_list(std::string const& str){
    std::vector<unsigned char> ret={};
    std::size_t count=0;
    char const* ptr=nullptr;
    for(std::size_t index=0;index<str.size();++index){
        if(count==0){
            ptr=&str[index];
        }
        if(str[index]=='0'||str[index]=='1'){
            ++count;
        }else if(str[index]==' '){
            ret.emplace_back(bstr_to_uchar(ptr,count));
            count=0;
        }else{
            throw std::runtime_error("bstr_to_uchar_list:error bchar");
        }
    }
    if(count!=0&&ptr!=nullptr){
        ret.emplace_back(bstr_to_uchar(ptr,count));
    }
    return ret;
}

int main(void){
    std::string const bstr="0010 0100 01 110 1111011 11 11111 010 000 0 001101 1010 111 100 0 001101 01111 000 001101 00 10 1 0 010 0 000 1 01111 10 11110 101011 1111101";
    std::vector<unsigned char> ret=bstr_to_uchar_list(bstr);

    std::cout<<"ret: [ ";
    for(auto num:ret){
        std::cout<<static_cast<long long>(num)<<" ";
    }
    std::cout<<"]\n";

    std::cout<<"ret str: ";
    for(unsigned char ch: ret){
        if(!std::isprint(ch)){
            std::cout<<"\\"<<(long long)ch<<' ';
        }else{
            std::cout<<static_cast<char>(ch)<<' ';
        }
    }
    std::cout<<'\n';

    return 0;
}
```

执行构建脚本生成可执行文件并运行,运行结果如下:
```bash
$ chmod +x build.sh
$ ./build.sh
ret: [ 2 4 1 6 123 3 31 2 0 0 13 10 7 4 0 13 15 0 13 0 2 1 0 2 0 0 1 15 2 30 43 125 ]
ret str: \2 \4 \1 \6 { \3 \31 \2 \0 \0 \13 \10 \7 \4 \0 \13 \15 \0 \13 \0 \2 \1 \0 \2 \0 \0 \1 \15 \2 \30 + } 
```
我现在开始怀疑这是摩斯密码(MORSE).

把`0`替换为`.`,把`1`替换为`-`,空格代表分隔符,这样空格,`0`,`1`都用上了.

将替换后的内容另存为`morse.txt`:
```
$ sed "s/0/./g;s/1/-/g" export/010.txt > morse.txt
$ cat morse.txt;echo
..-. .-.. .- --. ----.-- -- ----- .-. ... . ..--.- -.-. --- -.. . ..--.- .---- ... ..--.- .. -. - . .-. . ... - .---- -. ----. -.-.-- -----.-
$
```
使用`morse_cli.py`对`morse.txt`进行解密:
```bash
$ cat morse.txt | python3 ../../Tools/Crypto-morse_cli/morse_cli.py decode
FLAG[?----.--]M0RSE_CODE_1S_INTEREST1N9![?-----.-]
```
这里有未解析的`[?----.--]`和`[?-----.-]`.
将解密结果和之前的`solve.cpp`的运行结果中的`ret_str`信息一对比:
```
ret str: \2 \4 \1 \6 {           \3 \31 \2 \0 \0 \13 \10 \7 \4 \0 \13 \15 \0 \13 \0 \2 \1 \0 \2 \0 \0 \1 \15 \2 \30 + } 
         F  L  A  G  [?----.--]  M  0   R  S  E  _   C   O  D  E  _   1   S  _   I  N  T  E  R  E  S  T  1   N  9   ! [?-----.-]
```
会发现未解析的`[?----.--]`和`[?-----.-]`分别代表`{`和`}`的`ASCII`码表示.

至此我们得到了最终的`flag`,因为摩斯密码对字母不区分大小写,所以,
我们分别尝试提交一下
`flag{M0RSE_CODE_1S_INTEREST1N9!}`和
`flag{m0rse_code_1s_interest1n9!}`.

注意:将大写变换为小写的时候需要格外当心,这里有个坑:
```
flag{M0RSE_CODE_1S_INTEREST1N9!}
      ^         ^          ^ ^
    数字0     数字1    数字1 数字9
```

小写字母的`flag`提交通过,解答完毕.
