# Web-计算器-解题思路

打开`开发人员工具`,然后点击`</>元素`选项卡,
使用快捷键`ctrl-f`查找`flag`,发现没有找到.

接下来点击`源代码`选项卡,
之后查看`js/code.js`文件内容,使用快捷键`ctrl-f`查找`flag`,
找到`flag{ba43db9fdd1cfd4370edeff9204480fc}`,至此答题结束.

这里附上`js/code.js`的文件内容:
```javascript
$(function() {  
    var code = 9999; 
    function codes(){
    	
        var ranColor = '#' + ('00000' + (Math.random() * 0x1000000 << 0).toString(16)).slice(-6); //随机生成颜色
    	// alert(ranColor)
    	var ranColor2 = '#' + ('00000' + (Math.random() * 0x1000000 << 0).toString(16)).slice(-6); 
     	var num1 = Math.floor(Math.random() * 100);  
        var num2 = Math.floor(Math.random() * 100);  
        code = num1 + num2;  
        
        $("#code").html(num1 + "+" + num2 + "=?");  
        if ($("#code").hasClass("nocode")) {  
            $("#code").removeClass("nocode");  
            $("#code").addClass("code"); 
           
        }  
        $("#code").css('background',ranColor);
         $("#code").css('color',ranColor2);

    }
    codes()
   
    $("#code").on('click',codes)
      
    $("#check").click(function(){ 
        if ($(".input").val() == code && code != 9999) {  
            alert("flag{ba43db9fdd1cfd4370edeff9204480fc}");  
        } else {  
            alert("输入有误!");  
        }  
    });  
});  
```
